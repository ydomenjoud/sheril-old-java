// liste des offres
$offresListRes = mysql_query("SELECT * FROM z_galactique");
if (!$offresListRes) {
    echo mysql_error();
}
$offresList = [];
if($offresListRes && $row = mysql_fetch_assoc($offresListRes)) {
    $offresIdsList = explode(',', $row['ID_OFFRE']);
    $vendeursIdsList = explode(',', $row['VENDEUR']);
    $quantiteList = explode(',', $row['QUANTITE']);
    $codesList = explode(',', $row['CODE']);
    $prixList = explode(',', $row['PRIX']);
    for($i=0;$i<count($vendeursIdsList);$i++) {
        // les offres du commandant connecté sont affichées grisées, sans possibilité d'enchérir
        $offresList[] = [
            'VENDEUR' => $vendeursIdsList[$i],
            'QUANTITE' => $quantiteList[$i],
            'CODE' => $codesList[$i],
            'PRIX' => $prixList[$i],
            'ID_OFFRE' => $offresIdsList[$i],
            'PROPRE' => intval($vendeursIdsList[$i]) === intval($commandant),
        ];
    }
    // les offres du commandant connecté en premier, l'ordre d'origine étant conservé pour le reste
    $propres = [];
    $autres = [];
    foreach ($offresList as $offre) {
        if ($offre['PROPRE']) {
            $propres[] = $offre;
        } else {
            $autres[] = $offre;
        }
    }
    $offresList = array_merge($propres, $autres);
}
